CREATE OR REPLACE FUNCTION public.current_user_company_id()
RETURNS uuid
LANGUAGE sql
SECURITY DEFINER
SET search_path = public, pg_catalog
AS $$
  SELECT company_id
  FROM public.profiles
  WHERE id = auth.uid();
$$;

CREATE OR REPLACE FUNCTION public.current_user_role()
RETURNS text
LANGUAGE sql
SECURITY DEFINER
SET search_path = public, pg_catalog
AS $$
  SELECT role
  FROM public.profiles
  WHERE id = auth.uid();
$$;

CREATE OR REPLACE FUNCTION public.is_matelematics_admin()
RETURNS boolean
LANGUAGE sql
SECURITY DEFINER
SET search_path = public, pg_catalog
AS $$
  SELECT COALESCE(
    (SELECT role FROM public.profiles WHERE id = auth.uid()) = 'matelematics_admin',
    false
  );
$$;

CREATE OR REPLACE FUNCTION public.is_client_admin()
RETURNS boolean
LANGUAGE sql
SECURITY DEFINER
SET search_path = public, pg_catalog
AS $$
  SELECT COALESCE(
    (SELECT role FROM public.profiles WHERE id = auth.uid()) = 'client_admin',
    false
  );
$$;

CREATE OR REPLACE FUNCTION public.set_updated_at()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public, pg_catalog
AS $$
BEGIN
  IF TG_OP = 'UPDATE' THEN
    NEW.updated_at = now();
    RETURN NEW;
  END IF;

  IF TG_OP = 'INSERT' THEN
    NEW.updated_at = now();
    RETURN NEW;
  END IF;

  RETURN NEW;
END;
$$;

CREATE OR REPLACE FUNCTION public.protect_profile_security_fields()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public, pg_catalog
AS $$
DECLARE
  v_current_role text;
  v_current_company_id uuid;
BEGIN
  v_current_role := public.current_user_role();
  v_current_company_id := public.current_user_company_id();

  IF TG_OP = 'UPDATE' THEN
    -- user cannot change role or company_id
    IF v_current_role = 'user' THEN
      IF NEW.role IS DISTINCT FROM OLD.role THEN
        RAISE EXCEPTION 'user cannot modify profile role';
      END IF;

      IF NEW.company_id IS DISTINCT FROM OLD.company_id THEN
        RAISE EXCEPTION 'user cannot modify profile company_id';
      END IF;
    END IF;

    -- client_admin cannot change own company_id
    IF v_current_role = 'client_admin' THEN
      IF NEW.company_id IS DISTINCT FROM OLD.company_id THEN
        RAISE EXCEPTION 'client_admin cannot change company_id';
      END IF;
    END IF;

    -- client_admin cannot become or assign matelematics_admin
    IF v_current_role = 'client_admin' THEN
      IF NEW.role IS DISTINCT FROM OLD.role AND NEW.role = 'matelematics_admin' THEN
        RAISE EXCEPTION 'client_admin cannot become matelematics_admin';
      END IF;
    END IF;

    -- matelematics_admin keeps global management rights
    -- any other role handling is intentionally left to RLS and future review
  END IF;

  RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS profiles_protect_security_fields ON public.profiles;
CREATE TRIGGER profiles_protect_security_fields
BEFORE UPDATE ON public.profiles
FOR EACH ROW
EXECUTE FUNCTION public.protect_profile_security_fields();

IMPORTANT:
  1. profiles.company_id et profiles.role DOIVENT être correctement assignés
     à la création de chaque utilisateur. Les FKs et RLS échouent silencieusement
     si ces champs sont NULL ou incorrects.

     Mismatch = toutes les policies échouent.
  
  3. Role values MUST être exactement: 'user', 'client_admin', 'matelematics_admin'.
     Typos = RLS refuse tout accès.

MINEURS:
  1. devices.device_id nullable dans devices ET positions, créant des orphelins
     potentiels. Futur: ajouter CHECK (device_id IS NOT NULL).

     delete cascade manuel. Futur: revoir la stratégie de suppression.
  
  3. Pas d'audit trail pour les violations de policy (no RAISE NOTICE).
     Futur: ajouter logging audit si requis.
