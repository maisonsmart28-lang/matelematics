-- HELPER FUNCTIONS (05_functions.sql)
-- ============================================================

current_user_company_id() -> uuid
  Selector: SELECT company_id FROM profiles WHERE id = auth.uid()
  Security: SECURITY DEFINER, search_path = public, pg_catalog
  Grantee: authenticated role only
  Risk: None if profiles.company_id is correctly set at auth signup

current_user_role() -> text
  Selector: SELECT role FROM profiles WHERE id = auth.uid()
  Security: SECURITY DEFINER, search_path = public, pg_catalog
  Grantee: authenticated role only
  Risk: None if profiles.role is correctly initialized to 'user'

is_matelematics_admin() -> boolean
  Logic: role FROM profiles WHERE id = auth.uid() = 'matelematics_admin'
  Security: SECURITY DEFINER, safe default false
  Grantee: authenticated role only
  Risk: Requires real Supabase Auth assignment of matelematics_admin role

is_client_admin() -> boolean
  Logic: role FROM profiles WHERE id = auth.uid() = 'client_admin'
  Security: SECURITY DEFINER, safe default false
  Grantee: authenticated role only
  Risk: Requires real Supabase Auth assignment of client_admin role

set_updated_at() -> trigger
  Action: SET NEW.updated_at = now() on INSERT or UPDATE
  Applies to: drivers, devices, cameras
  Security: SECURITY DEFINER
  Risk: None (time-based maintenance only)

protect_profile_security_fields() -> trigger
  Action: BEFORE UPDATE ON profiles
  Logic:
    - user: cannot change role, cannot change company_id
    - client_admin: cannot change own company_id, cannot become matelematics_admin
    - matelematics_admin: no restrictions (can modify all fields)
  Security: SECURITY DEFINER
  Risk: Enforces tenure isolation and role hierarchy correctly
