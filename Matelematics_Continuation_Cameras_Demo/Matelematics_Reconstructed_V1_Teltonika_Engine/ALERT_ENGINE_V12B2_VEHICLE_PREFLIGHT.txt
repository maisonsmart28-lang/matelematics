function freshness(
  value:
    string | null | undefined,
) {
  if (!value) {
    return "Jamais";
  }

  const seconds =
    Math.max(
      0,
      Math.round(
        (
          Date.now() -
          new Date(
            value,
          ).getTime()
        ) /
        1000,
      ),
    );

  if (seconds < 10) {
    return "Ã€ l'instant";
  }

  if (seconds < 60) {
    return `Il y a ${seconds} s`;
  }

  const minutes =
    Math.floor(
      seconds / 60,
    );

  if (minutes < 60) {
    return `Il y a ${minutes} min`;
  }

  const hours =
    Math.floor(
      minutes / 60,
    );

  return `Il y a ${hours} h`;
}

async function authenticate(
  request:
    NextRequest,
) {
  const authorization =
    request.headers.get(
      "authorization",
    );


  const token =
    authorization
      ?.startsWith(
        "Bearer ",
      )
      ? authorization.slice(
          7,
        )
      : null;


  if (!token) {
    throw new Error(
      "AUTH_REQUIRED",
    );
  }


  const admin =
    getAdmin();


  const {
    data:
      userData,

    error:
      userError,
  } =
    await admin.auth.getUser(
      token,
    );


  if (
    userError ||
    !userData.user
  ) {
    throw new Error(
      "AUTH_REQUIRED",
    );
  }


  const {
    data:
      profile,

    error:
      profileError,
  } =
    await admin
      .from(
        "profiles",
      )
      .select(
        "id,role,company_id,partner_id",
      )

        "id",

      .single();


  if (
    profileError ||
    !profile
  ) {
    throw new Error(
      "PROFILE_REQUIRED",
    );
  }


  return {
    admin,

    profile:

  };

export async function GET(
  request:
    NextRequest,
) {
  try {
    const {
      admin,
      profile,
    } =
      await authenticate(
        request,
      );


    const companies =
      await getAccessibleCompanies(
        admin,
        profile,
      );


    if (
      companies.length ===
      0
    ) {
      return NextResponse.json({
        companies:
          [],

        selectedCompanyId:
          null,

        settings:
          [],

        canManage:
          canManage(
            profile,
          ),
      });
    }


    const requestedCompanyId =
      request.nextUrl
        .searchParams
        .get(
          "companyId",
        );


    const selectedCompanyId =
      requestedCompanyId &&
      isCompanyAccessible(
        companies,
        requestedCompanyId,
      )
        ? requestedCompanyId
        : companies[0].id;


    if (
      requestedCompanyId &&
      requestedCompanyId !==
        selectedCompanyId
    ) {
      throw new Error(
        "FORBIDDEN",
      );
    }


    const {
      data:
        rows,

      error:
        settingsError,
    } =
      await admin
        .from(
          "alert_settings",
        )
        .select(
          "rule_key,enabled,threshold_value,severity,updated_at",
        )
        .eq(
          "company_id",
          selectedCompanyId,
        )
        .is(
          "vehicle_id",
          null,
        );


    if (
      settingsError
    ) {
      throw settingsError;
    }


    const rowMap =
      new Map<
        string,
        {
          enabled:
            boolean;

          threshold_value:
            number |
            null;

          severity:
            string |
            null;

          updated_at:
            string |
            null;
        }
      >();


    for (
      const row of
      rows ??
      []
    ) {
      rowMap.set(
        row.rule_key,
        {
          enabled:
            row.enabled,

          threshold_value:
            row.threshold_value ===
              null
              ? null
              : Number(
                  row.threshold_value,
                ),

          severity:
            row.severity,

          updated_at:
            row.updated_at,
        },
      );
    }


    const settings =
      RULES.map(
        (
          rule,
        ) => {
          const row =
            rowMap.get(
              rule.key,
            );


          const hasOverride =
            Boolean(
              row &&
              row.enabled !==
                false &&
              row.threshold_value !==
                null &&
              Number.isFinite(
                row.threshold_value,
              ),
            );


          return {
            ...rule,

            value:
              hasOverride
                ? row!
                    .threshold_value
                : rule.platformValue,

            source:
              hasOverride
                ? "company"
                : "platform",

            updatedAt:
              row?.updated_at ??
              null,
          };
        },
      );


    return NextResponse.json({
      companies,

      selectedCompanyId,

      settings,

      canManage:
        canManage(
          profile,
        ),
    });

  } catch (cause) {
    return errorResponse(
      cause,
    );
  }
}


export async function POST(
  request:
    NextRequest,
) {
  try {
    const {
      admin,
      profile,
    } =
      await authenticate(
        request,
      );


    if (
      !canManage(
        profile,
      )
    ) {
      throw new Error(
        "FORBIDDEN",
      );
    }


    const companies =
      await getAccessibleCompanies(
        admin,
        profile,
      );


    const body =
      (await request.json()) as {
        companyId?:
          unknown;

        action?:
          unknown;

        settings?:
          unknown;
      };


    const companyId =
      typeof body.companyId ===
        "string"
        ? body.companyId
        : "";


    if (
      !companyId ||
      !isCompanyAccessible(
        companies,
        companyId,
      )
    ) {
      throw new Error(
        "FORBIDDEN",
      );
    }


    if (
      body.action ===
      "reset"
    ) {
      const {
        error,
      } =
        await admin
          .from(
            "alert_settings",
          )
          .delete()
          .eq(
            "company_id",
            companyId,
          )
          .is(
            "vehicle_id",
            null,
          )
          .in(
            "rule_key",
            RULES.map(
              (
                rule,
              ) =>
                rule.key,
            ),
          );


      if (error) {
        throw error;
      }


      return NextResponse.json({
        ok:
          true,

        reset:
          true,
      });
    }


    if (
      !body.settings ||
      typeof body.settings !==
        "object" ||
      Array.isArray(
        body.settings,
      )
    ) {
      return NextResponse.json(
        {
          error:
            "INVALID_SETTINGS",
        },
        {
          status:
            400,
        },
      );
    }


    const incoming =
      body.settings as
        Record<
          string,
          unknown
        >;


    const validated:
      {
        key:
          RuleKey;

        value:
          number;
      }[] = [];


    for (
      const [
        rawKey,
        rawValue,
      ] of
      Object.entries(
        incoming,
      )
    ) {
      const key =
        rawKey as
          RuleKey;


      const definition =
        RULE_MAP.get(
          key,
        );


      if (
        !definition
      ) {
        return NextResponse.json(
          {
            error:
              `UNKNOWN_RULE:${rawKey}`,
          },
          {
            status:
              400,
          },
        );
      }


      const value =
        typeof rawValue ===
          "number"
          ? rawValue
          : Number(
              rawValue,
            );


      if (
        !Number.isFinite(
          value,
        ) ||
        value <
          definition.min ||
        value >
          definition.max
      ) {
        return NextResponse.json(
          {
            error:
              `INVALID_THRESHOLD:${rawKey}`,
          },
          {
            status:
              400,
          },
        );
      }


      validated.push({
        key,
        value,
      });
    }


    if (
      validated.length !==
      RULES.length
    ) {
      return NextResponse.json(
        {
          error:
            "ALL_RULES_REQUIRED",
        },
        {
          status:
            400,
        },
      );
    }


    /*
     * We intentionally avoid Supabase upsert here because
     * company-level uniqueness is implemented by a partial
     * unique index (vehicle_id IS NULL).
     *
     * Existing row -> UPDATE
     * Missing row  -> INSERT
     */
    for (
      const item of
      validated
    ) {
      const {
        data:
          existing,

        error:
          lookupError,
      } =
        await admin
          .from(
            "alert_settings",
          )
          .select(
            "id",
          )
          .eq(
            "company_id",
            companyId,
          )
          .is(
            "vehicle_id",
            null,
          )
          .eq(
            "rule_key",
            item.key,
          )
          .limit(
            1,
          )
          .maybeSingle();


      if (
        lookupError
      ) {
        throw lookupError;
      }


      if (
        existing
      ) {
        const {
          error:
            updateError,
        } =
          await admin
            .from(
              "alert_settings",
            )
            .update({
              enabled:
                true,

              threshold_value:
                item.value,
            })
            .eq(
              "id",
              existing.id,
            );


        if (
          updateError
        ) {
          throw updateError;
        }

      } else {
        const {
          error:
            insertError,
        } =
          await admin
            .from(
              "alert_settings",
            )
            .insert({
              company_id:
                companyId,

              vehicle_id:
                null,

              rule_key:
                item.key,

              enabled:
                true,

              threshold_value:
                item.value,
            });


        if (
          insertError
        ) {
          throw insertError;
        }
      }
    }


    return NextResponse.json({
      ok:
        true,
    });

  } catch (cause) {
    return errorResponse(
      cause,
    );
  }
}
